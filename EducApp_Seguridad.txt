<?php

namespace App\Http\Controllers;

use App\Http\Requests\AcudienteRequest;
use App\Services\AcudienteService;
use Illuminate\Http\JsonResponse;

class AcudienteController extends Controller
{
    protected AcudienteService $acudienteService;

    public function __construct(
        AcudienteService $acudienteService
    ) {
        $this->acudienteService = $acudienteService;
    }

    public function index(): JsonResponse
    {
        $acudientes = $this->acudienteService->getAll();

        return response()->json([
            'mensaje' => 'Acudientes obtenidos correctamente.',
            'datos' => $acudientes
        ], 200);
    }

    public function show(int $id): JsonResponse
    {
        $acudiente = $this->acudienteService->findById($id);

        if (!$acudiente) {
            return response()->json([
                'mensaje' => 'Acudiente no encontrado.'
            ], 404);
        }

        return response()->json([
            'mensaje' => 'Acudiente obtenido correctamente.',
            'datos' => $acudiente
        ], 200);
    }

    public function store(
        AcudienteRequest $request
    ): JsonResponse {
        $acudiente = $this->acudienteService->create(
            $request->validated()
        );

        return response()->json([
            'mensaje' => 'Acudiente creado correctamente.',
            'datos' => $acudiente
        ], 201);
    }

    public function update(
        AcudienteRequest $request,
        int $id
    ): JsonResponse {
        $acudiente = $this->acudienteService->update(
            $id,
            $request->validated()
        );

        if (!$acudiente) {
            return response()->json([
                'mensaje' => 'Acudiente no encontrado.'
            ], 404);
        }

        return response()->json([
            'mensaje' => 'Acudiente actualizado correctamente.',
            'datos' => $acudiente
        ], 200);
    }

    public function destroy(int $id): JsonResponse
    {
        $eliminado = $this->acudienteService->delete($id);

        if (!$eliminado) {
            return response()->json([
                'mensaje' => 'Acudiente no encontrado.'
            ], 404);
        }

        return response()->json([
            'mensaje' => 'Acudiente eliminado correctamente.'
        ], 200);
    }
}

<?php

namespace App\Http\Controllers\Concerns;

use Illuminate\Routing\Controllers\Middleware;

trait AuthorizesWithPermissions
{

    abstract public static function permissions(): array;

    public static function middleware(): array
    {
        $grouped = [];

        foreach (static::permissions() as $method => $permission) {
            $grouped[$permission][] = $method;
        }

        return collect($grouped)
            ->map(fn (array $methods, string $permission) => new Middleware(
                "permission:{$permission}",
                only: $methods,
            ))
            ->values()
            ->all();
    }
}

namespace App\Services;

use App\Interfaces\RolInterface;

use Illuminate\Database\Eloquent\Collection;

use Symfony\Component\HttpKernel\Exception\ConflictHttpException;

class RolService
{
    protected RolInterface $rolRepository;

    public function __construct(RolInterface $rolRepository)
    {
        $this->rolRepository = $rolRepository;
    }

    public function getAll(): Collection
    {
        return $this->rolRepository->getAll();
    }

    public function findById(int $id): ?Rol
    {
        return $this->rolRepository->findById($id);
    }

    public function create(array $data): Rol
    {
        if ($this->rolRepository->existsByTipo($data['Tipo'])) {
            throw ValidationException::withMessages([
                'Tipo' => [
                    'Este tipo de rol ya se encuentra registrado.'
                ]
            ]);
        }

        return $this->rolRepository->create($data);
    }

    public function update(int $id, array $data): ?Rol
    {
        $rol = $this->rolRepository->findById($id);

        if (!$rol) {
            return null;
        }

        if (
            isset($data['Tipo']) &&
            $this->rolRepository->existsByTipo(
                $data['Tipo'],
                $id
            )
        ) {
            throw ValidationException::withMessages([
                'Tipo' => [
                    'Ya existe otro rol con este tipo.'
                ]
            ]);
        }

        return $this->rolRepository->update($id, $data);
    }

    public function delete(int $id): bool
    {
        $rol = $this->rolRepository->findById($id);

        if (!$rol) {
            return false;
        }

        if ($this->rolRepository->hasUsuarios($id)) {
            throw new ConflictHttpException(
                'No se puede eliminar el rol porque tiene usuarios asociados.'
            );
        }

        return $this->rolRepository->delete($id);
    }
}

{
    protected UsuarioInterface $usuarioRepository;

    public function __construct(
        UsuarioInterface $usuarioRepository
    ) {
        $this->usuarioRepository = $usuarioRepository;
    }

    public function getAll(): Collection
    {
        return $this->usuarioRepository->getAll();
    }

    public function findById(int $id): ?Usuario
    {
        return $this->usuarioRepository->findById($id);
    }

    public function create(array $data): Usuario
    {
        $data = $this->normalizarDatos($data);

        $this->validarDocumento(
            $data['Tipo_Documento'],
            (string) $data['Numero_Documento']
        );

        if (!empty($data['Correo_Electronico'])) {
            $this->validarCorreo(
                $data['Correo_Electronico']
            );
        }

        return $this->usuarioRepository->create($data);
    }

    public function update(
        int $id,
        array $data
    ): ?Usuario {
        $usuario = $this->usuarioRepository->findById($id);

        if (!$usuario) {
            return null;
        }

        $data = $this->normalizarDatos($data);

        $tipoDocumento = $data['Tipo_Documento']
            ?? $usuario->Tipo_Documento;

        $numeroDocumento = $data['Numero_Documento']
            ?? $usuario->Numero_Documento;

        $this->validarDocumento(
            $tipoDocumento,
            (string) $numeroDocumento,
            $id
        );

        if (!empty($data['Correo_Electronico'])) {
            $this->validarCorreo(
                $data['Correo_Electronico'],
                $id
            );
        }

        return $this->usuarioRepository->update(
            $id,
            $data
        );
    }
